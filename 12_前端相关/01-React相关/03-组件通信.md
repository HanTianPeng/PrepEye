### 父传子通信-Props
  - 父组件传递数据，子组件标签身上绑定属性
  - 子组件接收数据，props的参数
  - 特殊的props，children
```tsx
// 父组件
function Parent(){
  const [userName, setUserName] = useState('张三');
  const [age, setAge] = useState(25);

  return (
    <Child 
      name={userName} 
      age={age} 
      isActive={true}
    />
  );
};

// 子组件：通过 props 接收
function Child(props){
    const {name, age, isActive} = props;
    return (
        <div>
          <p>姓名：{name}</p>
          <p>年龄：{age}</p>
          <p>状态：{isActive ? '活跃' : '未激活'}</p>
        </div>
    );
};
```

### 子传父通信-回调函数(Props传递函数)
```tsx
// 父组件：定义回调函数，通过 props 传给子组件
function Parent(){
  const [message, setMessage] = useState('');

  // 子组件调用的回调
  const handleChildMessage = (data: string) => {
    setMessage(data);
  };

  return (
    <div>
      <p>来自子组件的消息：{message}</p>
      <Child onSendMessage={handleChildMessage} />
    </div>
  );
};

interface ChildProps {
    onSendMessage: (data: string) => void;  // 定义回调函数类型
}

// 子组件：接收回调函数，在适当时机调用
function Child({onSendMessage}: ChildProps){
  const handleClick = () => {
    onSendMessage('你好，父组件！');
  };

  return <button onClick={handleClick}>发送消息给父组件</button>;
};
```

### 兄弟组件通信-共同父组件+状态提升
```jsx
import { useState } from 'react';

// ========== 父组件：状态提升到这里 ==========
function App() {
  // 唯一的数据源
  const [message, setMessage] = useState('');

  return (
    <div style={{ padding: '20px' }}>
      <h2>📨 兄弟组件通信</h2>
      
      {/* 兄弟A：输入框 */}
      <BrotherA message={message} onMessageChange={setMessage} />
      
      {/* 兄弟B：显示消息 */}
      <BrotherB message={message} />
    </div>
  );
}

// ========== 兄弟A：输入框 ==========
function BrotherA({ message, onMessageChange }) {
  return (
    <div style={{ marginBottom: '10px' }}>
      <label>✏️ 输入消息：</label>
      <input 
        value={message} 
        onChange={(e) => onMessageChange(e.target.value)} 
        placeholder="输入文字..."
      />
    </div>
  );
}

// ========== 兄弟B：显示消息 ==========
function BrotherB({ message }) {
  return (
    <div>
      <p>📄 兄弟B收到的消息：<strong>{message || '（暂无消息）'}</strong></p>
    </div>
  );
}

export default App;
```


### 爷孙组件通信-Context
  - 使用createContext方法创建一个上下文对象Ctx
  - 在顶层组件<App>中通过Ctx.Provider组件提供数据
  - 在底层组件<B>中通过useContext钩子函数获取消费数据
```tsx
// 1. 创建 Context
const UserContext = React.createContext<{
    user: { name: string; age: number } | null;
    setUser: (user: { name: string; age: number }) => void;
} | null>(null);

// 2. 父组件提供数据
const GrandParent = () => {
    const [user, setUser] = useState({ name: '赵六', age: 28 });

    return (
        <UserContext.Provider value={{ user, setUser }}>
            <Parent />
        </UserContext.Provider>
    );
};

// 3. 中间组件：直接透传，无需处理 props
const Parent = () => {
    return <Child />;
};

// 4. 深层的子孙组件：直接用 Context
const Child = () => {
    const context = useContext(UserContext);
    if (!context) return null;

    return (
        <div>
            <p>姓名：{context.user?.name}</p>
            <p>年龄：{context.user?.age}</p>
            <button onClick={() => context.setUser({ name: '孙七', age: 35 })}>
                修改用户
            </button>
        </div>
    );
};  
```
